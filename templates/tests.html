document.addEventListener("DOMContentLoaded", function() { const urlParams = new URLSearchParams(window.location.search); const pkg = urlParams.get('add_pkg'); if (pkg) { let testsToAdd = []; if(pkg === 'executive') testsToAdd = ['cbc', 'thyroid', 'liver', 'fasting']; if(pkg === 'diabetic') testsToAdd = ['hba1c', 'fasting', 'lipid', 'urine']; if(pkg === 'pcod') testsToAdd = ['fsh', 'testosterone', 'insulin', 'thyroid']; const cards = document.querySelectorAll('.test-card'); cards.forEach(card => { const testName = card.dataset.name.toLowerCase(); testsToAdd.forEach(keyword => { if(testName.includes(keyword)) { card.querySelector('.btn-add').click(); } }); }); // Open cart automatically setTimeout(() => toggleCart(), 500); } });
